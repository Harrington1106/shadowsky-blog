<%*
/*
 * 新文章 —— 一键生成一篇符合 shadowquake.top 发布要求的草稿。
 *
 * 只问「人才知道」的四件事：标题 / slug / 分类 / 标签（外加可选封面）。
 * excerpt、readTime、lastModified 一律不写 —— 那三个是机械字段，
 * 由 publish-post.mjs 在发布时算。手写它们是上一版工作流最容易出错的地方
 * （现有 19 篇的 excerpt 几乎全是把标题又抄了一遍）。
 *
 * 生成的文件名就是线上地址：/post/<日期>-<slug>
 */

// 这个模板只用于「新建」。插到写了一半的文章里会把它重命名掉，所以先挡住。
if ((tp.file.content || '').trim()) {
    throw new Error('当前文件不是空的 —— 这个模板只用于新建文章。先 Ctrl+N 建一个空白笔记再运行。');
}

/*
 * 线上的 frontmatter 解析器（lib/posts.js）是手写的：按第一个 ':' 切、
 * 只剥首尾各一个引号。所以每个值必须单行、不能含 ---，
 * 而且值里的半角引号只会被原样吐出来（写 \" 会在页面上显示成 \"），
 * 这里统一换成中文引号。
 */
const clean = (s) => String(s ?? '')
    .replace(/[\r\n]+/g, ' ')
    .replace(/-{3,}/g, '—')
    .replace(/"/g, '”')
    .trim();

/** slug 就是线上地址的一部分，只允许小写英文、数字、连字符 */
const slugify = (s) => String(s ?? '').toLowerCase().trim()
    .replace(/[^a-z0-9]+/g, '-')
    .replace(/^-+|-+$/g, '')
    .replace(/-{2,}/g, '-');

/*
 * 分类清单和「这个 slug 是不是已经被占了」都从 posts/ 现读，
 * 免得清单在这儿写死、线上加了新分类却不知道。
 * posts/ 就在库里，所以直接走 Obsidian 自己的 API —— 比 require('fs') 稳，
 * 也不依赖桌面端。拿不到就退回下面这份固定清单，读不到不该拦着人写文章。
 */
let categories = ['教程', '技术', '博客运维', '前端', '软件推荐', '效率工具'];
const takenSlugs = new Set();
try {
    const found = new Set();
    for (const f of app.vault.getMarkdownFiles()) {
        if (!f.path.startsWith('posts/')) continue;
        takenSlugs.add(f.basename);
        const head = (await app.vault.cachedRead(f)).slice(0, 1000);
        const m = head.match(/^category:\s*"?([^"\n]+)"?\s*$/m);
        const c = m && m[1].trim();
        if (c && c !== '未分类') found.add(c);
    }
    if (found.size) categories = [...found].sort();
} catch (e) {
    // 读不到就算了，用上面的固定清单
}

const date = tp.date.now('YYYY-MM-DD');

// ── 1. 标题 ──
const title = clean(await tp.system.prompt('文章标题（中文，会显示在页面顶部）', '', true));
if (!title) throw new Error('没填标题，已取消');

// ── 2. slug ──（发布后再改等于换地址、丢外链，所以这里逐条校验）
let slug = '';
let tip = '';
for (;;) {
    const ans = await tp.system.prompt(
        `${tip}英文 slug —— 线上地址 /post/${date}-<slug>，发布后不能改`,
        slugify(title), true,
    );
    slug = slugify(ans);
    if (!slug) { tip = '⚠ 只能用英文字母 / 数字 / 连字符 —— '; continue; }
    if (takenSlugs.has(`${date}-${slug}`)) { tip = `⚠ ${date}-${slug} 线上已有同名文章 —— `; continue; }
    if (app.vault.getAbstractFileByPath(`drafts/${date}-${slug}.md`)) { tip = '⚠ 草稿箱里已经有这篇 —— '; continue; }
    break;
}

// ── 3. 分类 ──
const NEW_CATEGORY = '＋ 新分类…';
let category = await tp.system.suggester(
    [...categories, NEW_CATEGORY], [...categories, NEW_CATEGORY], true, '选一个分类',
);
if (category === NEW_CATEGORY) category = clean(await tp.system.prompt('新分类名', '', true));
if (!category) throw new Error('没选分类，已取消');

// ── 4. 标签 ──（可留空；线上解析器对 tags 直接 JSON.parse，必须是单行合法数组）
const tagsRaw = await tp.system.prompt('标签（逗号分隔，可留空）', '') || '';
const tags = tagsRaw.split(/[,，]/).map(clean).filter(Boolean);

// ── 5. 封面 ──（可留空 → 落到分类默认图；PicList 的跨境地址发布时会自动镜像到自家服务器）
const cover = clean(await tp.system.prompt('封面图地址（可留空）', '') || '');

/*
 * 文件名 = 线上 slug，并且**必须落在 drafts/ 里** ——
 * 库根是 content/，而发布台只列 drafts/ 顶层的 *.md，落在别处就发不出去。
 * 用 move 而不是 rename：新笔记是在库根创建的，要连目录一起换。
 */
await tp.file.move(`drafts/${date}-${slug}`);
-%>
---
title: "<% title %>"
date: "<% date %>"
category: "<% category %>"
author: "Thoi"
tags: <% JSON.stringify(tags) %>
coverImage: "<% cover %>"
---

<% tp.file.cursor() %>
